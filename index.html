-- السماح للموظف بقراءة مهام قطاعه drop policy if exists "employees can read sector tasks" on public.tasks; create policy "employees can read sector tasks" on public.tasks for select to authenticated using ( exists ( select 1 from public.users u where u.id = auth.uid() and u.is_active = true and ( public.tasks.owner_sector_id = u.sector_id or public.tasks.target_sector_id = u.sector_id or public.tasks.primary_responsible_id in ( select u2.id from public.users u2 where u2.sector_id = u.sector_id and u2.is_active = true ) or exists ( select 1 from public.task_assignees ta join public.users au on au.id = ta.user_id where ta.task_id = public.tasks.id and au.sector_id = u.sector_id and au.is_active = true ) ) ) ); -- السماح للموظف بقراءة إسنادات مهام قطاعه drop policy if exists "employees can read sector task assignees" on public.task_assignees; create policy "employees can read sector task assignees" on public.task_assignees for select to authenticated using ( exists ( select 1 from public.users me join public.users assignee_user on assignee_user.id = public.task_assignees.user_id where me.id = auth.uid() and me.is_active = true and assignee_user.sector_id = me.sector_id and assignee_user.is_active = true ) or exists ( select 1 from public.users me join public.tasks t on t.id = public.task_assignees.task_id where me.id = auth.uid() and me.is_active = true and ( t.owner_sector_id = me.sector_id or t.target_sector_id = me.sector_id ) ) ); -- السماح للموظف بقراءة تعليقات مهام قطاعه drop policy if exists "employees can read sector task comments" on public.task_comments; create policy "employees can read sector task comments" on public.task_comments for select to authenticated using ( exists ( select 1 from public.users me join public.tasks t on t.id = public.task_comments.task_id where me.id = auth.uid() and me.is_active = true and ( t.owner_sector_id = me.sector_id or t.target_sector_id = me.sector_id or t.primary_responsible_id in ( select u2.id from public.users u2 where u2.sector_id = me.sector_id and u2.is_active = true ) or exists ( select 1 from public.task_assignees ta join public.users au on au.id = ta.user_id where ta.task_id = t.id and au.sector_id = me.sector_id and au.is_active = true ) ) ) );
